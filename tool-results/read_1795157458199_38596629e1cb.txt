    11→// Sanitize text for WinAnsi-encoded standard fonts (replace Unicode chars
    12→// that Helvetica/Helvetica-Bold cannot encode with ASCII equivalents).
    13→function sanitize(text: string): string {
    14→  return text
    15→    .replace(/≥/g, '>=').replace(/≤/g, '<=').replace(/→/g, '->').replace(/←/g, '<-')
    16→    .replace(/—/g, '-').replace(/–/g, '-').replace(/'/g, "'").replace(/'/g, "'")
    17→    .replace(/"/g, '"').replace(/"/g, '"').replace(/•/g, '*').replace(/·/g, '-')
    18→    .replace(/&bull;/g, '*').replace(/&middot;/g, '-').replace(/&mdash;/g, '-')
    19→    .replace(/&ndash;/g, '-').replace(/&rarr;/g, '->').replace(/&ldquo;/g, '"')
    20→    .replace(/&rdquo;/g, '"').replace(/&lsquo;/g, "'").replace(/&rsquo;/g, "'")
    21→    .replace(/&amp;/g, '&').replace(/&lt;/g, '<').replace(/&gt;/g, '>')
    22→    .replace(/\u00A0/g, ' ').replace(/\u2026/g, '...')
    23→}
    24→
    25→function hex(h: string): ReturnType<typeof rgb> {
    26→  const r = parseInt(h.slice(1, 3), 16) / 255
    27→  const g = parseInt(h.slice(3, 5), 16) / 255
    28→  const b = parseInt(h.slice(5, 7), 16) / 255
    29→  return rgb(r, g, b)
    30→}

    32→const C = {
    33→  NAVY: hex('#0F2A43'), TEAL: hex('#0E7C66'), AMBER: hex('#B45309'),
    34→  SLATE: hex('#334155'), MUTED: hex('#64748B'), LIGHT: hex('#F1F5F9'),
    35→  BORDER: hex('#CBD5E1'), GREEN: hex('#15803D'), WHITE: rgb(1, 1, 1),
    36→  BG_LIGHT: hex('#FAFAFA'), EMERALD_BG: hex('#ECFDF5'),
    37→}

    39→function lvlColor(level: string) {
    40→  if (['High', 'Dominant'].includes(level)) return C.GREEN
    41→  if (['Moderate'].includes(level)) return C.TEAL
    42→  return C.AMBER
    43→}

    45→// Wrap text to fit a max width given a font + size. Returns array of lines.
    46→function wrapText(text: string, font: PDFFont, size: number, maxWidth: number): string[] {
    47→  text = sanitize(text)
    48→  const words = text.split(/\s+/)
    49→  const lines: string[] = []
    50→  let line = ''
    51→  for (const word of words) {
    52→    const test = line ? `${line} ${word}` : word
    53→    if (font.widthOfTextAtSize(test, size) <= maxWidth) {
    54→      line = test
    55→    } else {
    56→      if (line) lines.push(line)
    57→      line = word
    58→    }
    59→  }
    60→  if (line) lines.push(line)
    61→  return lines
    62→}

   291→  // Radar/spider chart — visual element for dimension profiles
   292→  function drawRadarChart(chartDims: any[], cx: number, cy: number, radius: number) {
   293→    const n = chartDims.length
   294→    if (n < 3) return
   295→    // Grid: concentric polygons at 25%, 50%, 75%, 100%
   296→    for (const pct of [0.25, 0.5, 0.75, 1.0]) {
   297→      const gridColor = pct === 1.0 ? C.BORDER : hex('#E2E8F0')
   298→      for (let i = 0; i < n; i++) {
   299→        const a1 = (2 * Math.PI * i / n) - Math.PI / 2
   300→        const a2 = (2 * Math.PI * (i + 1) / n) - Math.PI / 2
   301→        const r = radius * pct
   302→        page.drawLine({
   303→          start: { x: cx + r * Math.cos(a1), y: cy + r * Math.sin(a1) },
   304→          end: { x: cx + r * Math.cos(a2), y: cy + r * Math.sin(a2) },
   305→          thickness: 0.5, color: gridColor,
   306→        })
   307→      }
   308→    }
   309→    // Spokes
   310→    for (let i = 0; i < n; i++) {
   311→      const a = (2 * Math.PI * i / n) - Math.PI / 2
   312→      page.drawLine({
   313→        start: { x: cx, y: cy },
   314→        end: { x: cx + radius * Math.cos(a), y: cy + radius * Math.sin(a) },
   315→        thickness: 0.4, color: hex('#E2E8F0'),
   316→      })
   317→    }
   318→    // Data polygon points
   319→    const pts = chartDims.map((d, i) => {
   320→      const a = (2 * Math.PI * i / n) - Math.PI / 2
   321→      const r = radius * Math.min(1, (d.pct || 0) / 100)
   322→      return { x: cx + r * Math.cos(a), y: cy + r * Math.sin(a) }
   323→    })
   324→    // Filled polygon — draw lines from center to each vertex to create a "filled" look
   325→    for (let i = 0; i < n; i++) {
   326→      page.drawLine({ start: { x: cx, y: cy }, end: pts[i], thickness: 1, color: hex('#0E7C66'), opacity: 0.2 })
   327→    }
   328→    // Outline (thicker)
   329→    for (let i = 0; i < n; i++) {
   330→      const p1 = pts[i], p2 = pts[(i + 1) % n]
   331→      page.drawLine({ start: p1, end: p2, thickness: 2, color: C.TEAL })
   332→    }
   333→    // Data points
   334→    pts.forEach((p) => { page.drawCircle({ x: p.x, y: p.y, size: 3, color: C.TEAL }) })
   335→    // Labels
   336→    chartDims.forEach((d, i) => {
   337→      const a = (2 * Math.PI * i / n) - Math.PI / 2
   338→      const lr = radius + 16
   339→      const lx = cx + lr * Math.cos(a), ly = cy + lr * Math.sin(a)
   340→      const name = sanitize((d.name || d.code || '').slice(0, 14))
   341→      const tw = font.widthOfTextAtSize(name, 7)
   342→      page.drawText(name, { x: lx - tw / 2, y: ly - 3, size: 7, font, color: C.SLATE })
   343→    })
   344→  }

   740→function fmtDate(s?: string): string {
   741→  if (!s) return '—'
   742→  try { return new Date(s.replace('Z', '')).toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' }) }
   743→  catch { return String(s) }
   744→}

   747→export interface ManualContent {
   748→  title: string; purpose: string; constructsMeasured: string; targetPopulation: string
   749→  scoringMethodology: string; behaviouralIndicators: string[]; strengthsAndDevelopment: string
   750→  counsellingApproach: string; limitations: string; sampleInterpretation: string
   751→  dimensions: { name: string; description: string; high: string; moderate: string; low: string }[]

   754→export async function generateManualPdf(content: ManualContent, id: string): Promise<Buffer> {
   755→  const doc = await PDFDocument.create()
   756→  const font = await doc.embedFont(StandardFonts.Helvetica)
   757→  const fontBold = await doc.embedFont(StandardFonts.HelveticaBold)
   758→  let page = doc.addPage([PAGE_W, PAGE_H])
   759→  const PLUM = hex('#5B21B6')
   760→  let pn = 1
   761→
   762→  function drawHF(p: PDFPage) {
   763→    p.drawRectangle({ x: 0, y: PAGE_H - 40, width: PAGE_W, height: 40, color: PLUM })
   764→    p.drawText('Counsellor Interpretation Manual', { x: MARGIN, y: PAGE_H - 20, size: 10, font: fontBold, color: C.WHITE })
   765→    p.drawText('CONFIDENTIAL', { x: PAGE_W - MARGIN - 70, y: PAGE_H - 20, size: 8, font, color: C.WHITE })
   766→    p.drawLine({ start: { x: MARGIN, y: 25 }, end: { x: PAGE_W - MARGIN, y: 25 }, thickness: 0.5, color: C.BORDER })
   767→    p.drawText(`Page ${pn}`, { x: PAGE_W - MARGIN - 40, y: 18, size: 7.5, font, color: C.MUTED })
   768→  }
   769→  drawHF(page)
   770→
   771→  let y = PAGE_H - 55
   772→  function newPage() { page = doc.addPage([PAGE_W, PAGE_H]); pn++; drawHF(page); y = PAGE_H - 55 }
   773→  function ensureSpace(h: number) { if (y - h < 45) newPage() }
   774→  function heading(text: string, size = 13, color = PLUM) { ensureSpace(size + 8); page.drawText(text, { x: MARGIN, y: y - size, size, font: fontBold, color }); y -= size + 6 }
   775→  function body(text: string, size = 9.5) {
   776→    const lines = wrapText(text, font, size, contentW)
   777→    for (const l of lines) { ensureSpace(size + 2); page.drawText(l, { x: MARGIN, y: y - size, size, font, color: C.SLATE }); y -= size + 2 }
   778→  }
   779→  const contentW = PAGE_W - 2 * MARGIN
   780→
   781→  heading(content.title, 17, PLUM); y -= 4
   782→  const sections: [string, string][] = [
   783→    ['Purpose', content.purpose], ['Constructs Measured', content.constructsMeasured],
   784→    ['Target Population', content.targetPopulation], ['Scoring Methodology', content.scoringMethodology],
   785→  ]
   786→  sections.forEach(([h, b]) => { heading(h); body(b) })
   787→
   788→  if (content.dimensions?.length) {
   789→    heading('Dimensions & Interpretation Bands')
   790→    content.dimensions.forEach((d) => {
   791→      ensureSpace(60)
   792→      page.drawText(d.name, { x: MARGIN, y: y - 10, size: 9.5, font: fontBold, color: C.SLATE }); y -= 12
   793→      body(d.description.slice(0, 150), 8.5)
   794→      const bands: [string, string, any][] = [['High: ', d.high, C.GREEN], ['Moderate: ', d.moderate, C.TEAL], ['Low: ', d.low, C.AMBER]]
   795→      bands.forEach(([label, val, col]) => {
   796→        const lines = wrapText(`${label}${val.slice(0, 150)}`, font, 8, contentW)
   797→        lines.forEach((l) => { ensureSpace(11); page.drawText(l, { x: MARGIN, y: y - 8, size: 8, font: fontBold, color: col }); y -= 11 })
   798→      })
   799→      y -= 4
   800→    })
   801→  }
   802→
   803→  heading('Behavioural Indicators')
   804→  content.behaviouralIndicators?.forEach((b) => {
   805→    const lines = wrapText(`• ${b}`, font, 9.5, contentW - 10)
   806→    lines.forEach((l) => { ensureSpace(12); page.drawText(l, { x: MARGIN + 10, y: y - 10, size: 9.5, font, color: C.SLATE }); y -= 12 })
   807→  })
   808→
   809→  const more: [string, string][] = [
   810→    ['Strengths & Development Areas', content.strengthsAndDevelopment],
   811→    ['Counselling Approach', content.counsellingApproach],
   812→    ['Assessment Limitations', content.limitations],
   813→    ['Sample Interpretation', content.sampleInterpretation],
   814→  ]
   815→  more.forEach(([h, b]) => { heading(h); body(b) })
   816→
   817→  doc.setTitle(content.title)
   818→  doc.setAuthor('Psychometric Testing Platform')
   819→
   820→  const pdfBytes = await doc.save()
   821→  return Buffer.from(pdfBytes)
   822→}
