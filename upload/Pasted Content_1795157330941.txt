3. Shared Scoring Engine (src/lib/tests/commerce/scoring.ts)

import { CLUSTERS, FINANCIAL_CLUSTERS, QUESTIONS } from './data';
import { ClusterResult, CommerceReportResult, InterestLevel, StudentInfo } from './types';

export function calculateCommerceResults(
  student: StudentInfo,
  answers: Array<number | null>
): CommerceReportResult {
  const rawMap: Record<string, number> = {};
  Object.keys(CLUSTERS).forEach((c) => {
    rawMap[c] = 0;
  });

  let answeredCount = 0;
  QUESTIONS.forEach((q, idx) => {
    const val = answers[idx];
    if (val !== null && val !== undefined && val >= 1 && val <= 5) {
      rawMap[q.cluster] += val;
      answeredCount++;
    }
  });

  const sortedEntries = Object.entries(rawMap).sort((a, b) => b[1] - a[1]);
  const third = Math.floor(sortedEntries.length / 3); // 6 High, 6 Medium, 7 Low

  const rankMap: Record<string, number> = {};
  const levelMap: Record<string, InterestLevel> = {};

  sortedEntries.forEach(([cluster], idx) => {
    rankMap[cluster] = idx + 1;
    levelMap[cluster] = idx < third ? 'High' : idx < third * 2 ? 'Medium' : 'Low';
  });

  let financialRaw = 0;
  let nonFinancialRaw = 0;
  sortedEntries.forEach(([cluster, score]) => {
    if (FINANCIAL_CLUSTERS.has(cluster)) financialRaw += score;
    else nonFinancialRaw += score;
  });

  const totalRawScore = financialRaw + nonFinancialRaw;
  const denom = totalRawScore || 1;
  const financialPct = Math.round((100 * financialRaw) / denom);
  const nonFinancialPct = 100 - financialPct;
  const preferredDomain =
    financialRaw >= nonFinancialRaw ? 'Financial Domain' : 'Non-Financial Domain';

  const toClusterResult = (cluster: string): ClusterResult => ({
    cluster,
    domain: FINANCIAL_CLUSTERS.has(cluster) ? 'Financial' : 'Non-Financial',
    rawScore: rawMap[cluster],
    percentage: Math.round((rawMap[cluster] / 20) * 100),
    level: levelMap[cluster],
    rank: rankMap[cluster],
    roles: CLUSTERS[cluster],
  });

  const sortedClusters = sortedEntries.map(([c]) => toClusterResult(c));
  const allClustersInOrder = Object.keys(CLUSTERS).map(toClusterResult);

  const now = new Date();
  const dateFormatted = now.toLocaleDateString('en-GB', {
    day: 'numeric',
    month: 'long',
    year: 'numeric',
  });

  return {
    student,
    dateFormatted,
    reportId: `CCS-${now.getFullYear()}-${String(QUESTIONS.length).padStart(3, '0')}`,
    totalQuestions: QUESTIONS.length,
    answeredCount,
    totalRawScore,
    overallIndexPct: Math.round((totalRawScore / (19 * 20)) * 100),
    financialRaw,
    nonFinancialRaw,
    financialPct,
    nonFinancialPct,
    preferredDomain,
    highCount: sortedClusters.filter((c) => c.level === 'High').length,
    mediumCount: sortedClusters.filter((c) => c.level === 'Medium').length,
    lowCount: sortedClusters.filter((c) => c.level === 'Low').length,
    sortedClusters,
    allClustersInOrder,
    top8: sortedClusters.slice(0, 8),
    top5: sortedClusters.slice(0, 5),
    top3: sortedClusters.slice(0, 3),
    financialClusters: sortedClusters.filter((c) => c.domain === 'Financial'),
    nonFinancialClusters: sortedClusters.filter((c) => c.domain === 'Non-Financial'),
  };
}

const W = PageSizes.A4[0]; // 595.28
const H = PageSizes.A4[1]; // 841.89
const M = 36;              // Margin
const CW = W - M * 2;      // Content width (523.28)

// Sanitize strings for WinAnsi standard fonts
function clean(str: string): string {
  return str
    .replace(/[\u2013\u2014]/g, '-')
    .replace(/[\u2018\u2019]/g, "'")
    .replace(/[\u201C\u201D]/g, '"')
    .replace(/[^\x20-\x7E]/g, ' ');
}

function wrapText(text: string, font: PDFFont, size: number, maxWidth: number): string[] {
  const words = clean(text).split(/\s+/);
  const lines: string[] = [];
  let line = '';
  for (const w of words) {
    const test = line ? `${line} ${w}` : w;
    if (font.widthOfTextAtSize(test, size) <= maxWidth) {
      line = test;
    } else {
      if (line) lines.push(line);
      line = w;
    }
  }
  if (line) lines.push(line);
  return lines;
}

function levelColors(level: InterestLevel) {
  if (level === 'High') return { fg: C.greenDark, bg: C.greenBg, bar: C.green };
  if (level === 'Medium') return { fg: C.amberDark, bg: C.amberBg, bar: C.amber };
  return { fg: C.roseDark, bg: C.roseBg, bar: C.rose };
}

export async function generateCommercePdf(
  report: CommerceReportResult,
  branding: BrandingConfig = {}
): Promise<Uint8Array> {
  const pdf = await PDFDocument.create();
  const font = await pdf.embedFont(StandardFonts.Helvetica);
  const fontBold = await pdf.embedFont(StandardFonts.HelveticaBold);
  const serifBold = await pdf.embedFont(StandardFonts.TimesRomanBold);

  const clientTitle = clean(branding.clientName || 'Commerce Career Selector');
  const totalPages = 10;

  // Helper: Draw Sub-page Header & Footer
  const initPage = (pageNum: number, sectionTitle?: string): { page: PDFPage; y: number } => {
    const page = pdf.addPage(PageSizes.A4);

    // Top colorful accent strip
    page.drawRectangle({ x: 0, y: H - 8, width: W * 0.6, height: 8, color: C.blue });
    page.drawRectangle({ x: W * 0.6, y: H - 8, width: W * 0.4, height: 8, color: C.green });

    // Footer
    page.drawLine({ start: { x: M, y: 38 }, end: { x: W - M, y: 38 }, thickness: 0.75, color: C.border });
    page.drawText(clean(`${clientTitle} - Candidate: ${report.student.name}`), {
      x: M, y: 24, size: 8, font, color: C.muted,
    });
    const pgLabel = `Page ${pageNum} of ${totalPages}`;
    page.drawText(pgLabel, {
      x: W - M - fontBold.widthOfTextAtSize(pgLabel, 8), y: 24, size: 8, font: fontBold, color: C.navy,
    });

    if (!sectionTitle) {
      return { page, y: H - 32 };
    }

    page.drawText(clean(`${clientTitle.toUpperCase()}  |  COMMERCE CAREER SELECTOR`), {
      x: M, y: H - 30, size: 8, font: fontBold, color: C.blue,
    });
    page.drawText(clean(sectionTitle), {
      x: M, y: H - 50, size: 16, font: serifBold, color: C.navy,
    });
    page.drawText(clean(`Candidate: ${report.student.name}   |   Phone: ${report.student.phone}   |   Date: ${report.dateFormatted}`), {
      x: M, y: H - 64, size: 8, font, color: C.muted,
    });
    page.drawLine({ start: { x: M, y: H - 72 }, end: { x: W - M, y: H - 72 }, thickness: 1.5, color: C.blue });

    return { page, y: H - 92 };
  };

  // Helper: Draw Section Heading with Left Accent Bar
  const drawHeading = (page: PDFPage, y: number, title: string): number => {
    page.drawRectangle({ x: M, y: y - 3, width: 4, height: 14, color: C.blue });
    page.drawText(clean(title), { x: M + 10, y, size: 12, font: serifBold, color: C.navy });
    return y - 18;
  };

  // Helper: Draw Paragraph
  const drawPara = (page: PDFPage, y: number, text: string, size = 9.2, maxWidth = CW, x = M): number => {
    const lines = wrapText(text, font, size, maxWidth);
    for (const line of lines) {
      page.drawText(line, { x, y, size, font, color: C.slate });
      y -= size * 1.45;
    }
    return y - 4;
  };

  // Helper: Draw Badge
  const drawBadge = (page: PDFPage, x: number, y: number, level: InterestLevel) => {
    const col = levelColors(level);
    const label = level.toUpperCase();
    const bw = 48;
    page.drawRectangle({ x, y: y - 3, width: bw, height: 13, color: col.bg, borderColor: col.fg, borderWidth: 0.7 });
    const tw = fontBold.widthOfTextAtSize(label, 7);
    page.drawText(label, { x: x + (bw - tw) / 2, y: y + 1, size: 7, font: fontBold, color: col.fg });
  };

  // Helper: Draw Insight Callout Box
  const drawInsight = (page: PDFPage, y: number, title: string, body: string): number => {
    const lines = wrapText(body, font, 8.6, CW - 24);
    const boxH = 26 + lines.length * 12.5;
    page.drawRectangle({ x: M, y: y - boxH, width: CW, height: boxH, color: C.softBlue, borderColor: C.blue, borderWidth: 0.6 });
    page.drawRectangle({ x: M, y: y - boxH, width: 4, height: boxH, color: C.blue });
    page.drawText(clean(title), { x: M + 12, y: y - 14, size: 9.5, font: fontBold, color: C.blue });
    let ty = y - 28;
    for (const l of lines) {
      page.drawText(l, { x: M + 12, y: ty, size: 8.6, font, color: C.slate });
      ty -= 12.5;
    }
    return y - boxH - 12;
  };

  // Helper: Draw Horizontal Bar Chart
  const drawHorizBars = (
    page: PDFPage,
    y: number,
    items: Array<{ label: string; pct: number; sub?: string; level?: InterestLevel; color?: ReturnType<typeof rgb> }>,
    barHeight = 9,
    rowGap = 24
  ): number => {
    const colors = [C.blue, C.sky, C.green, C.indigo, C.amber];
    items.forEach((item, idx) => {
      const col = item.color || (item.level ? levelColors(item.level).bar : colors[idx % colors.length]);
      page.drawText(clean(item.label), { x: M + 8, y, size: 8.5, font: fontBold, color: C.navy });
      const pctStr = item.sub ? `${item.pct}% (${item.sub})` : `${item.pct}%`;
      const rightOffset = item.level ? 62 : 8;
      page.drawText(clean(pctStr), {
        x: W - M - rightOffset - fontBold.widthOfTextAtSize(pctStr, 8.2),
        y, size: 8.2, font: fontBold, color: C.slate,
      });
      if (item.level) drawBadge(page, W - M - 56, y, item.level);

      const trackY = y - barHeight - 3;
      const trackW = CW - 16;
      page.drawRectangle({ x: M + 8, y: trackY, width: trackW, height: barHeight, color: C.border });
      page.drawRectangle({ x: M + 8, y: trackY, width: Math.max(2, (trackW * item.pct) / 100), height: barHeight, color: col });
      y -= rowGap;
    });
    return y;
  };

  // Helper: Draw Cluster Table (Pages 7-10)
  const drawClusterTable = (page: PDFPage, startY: number, clusters: ClusterResult[]): number => {
    let y = startY;
    const colW = [132, 92, 58, CW - 282];

5. Backend API Route (src/app/api/reports/commerce-pdf/route.ts)
TypeScript
import { NextRequest, NextResponse } from 'next/server';
import { calculateCommerceResults } from '@/lib/tests/commerce/scoring';
import { generateCommercePdf } from '@/lib/tests/commerce/pdfGenerator';
import { BrandingConfig, StudentInfo } from '@/lib/tests/commerce/types';

export async function POST(req: NextRequest) {
  try {
    const body = (await req.json()) as {
      student: StudentInfo;
      answers: Array<number | null>;
      branding?: BrandingConfig;
    };

    if (!body?.student?.name || !Array.isArray(body?.answers)) {
      return NextResponse.json({ error: 'Invalid payload: student and answers are required.' }, { status: 400 });
    }

    const report = calculateCommerceResults(body.student, body.answers);
    const pdfBytes = await generateCommercePdf(report, body.branding);

    const safeName = body.student.name.replace(/[^a-zA-Z0-9_-]/g, '_');

    return new NextResponse(Buffer.from(pdfBytes), {
      status: 200,
      headers: {
        'Content-Type': 'application/pdf',
        'Content-Disposition': `attachment; filename="Commerce_Career_Report_${safeName}.pdf"`,
        'Cache-Control': 'no-store',
      },
    });
  } catch (err) {
    console.error('PDF generation failed:', err);
    return NextResponse.json({ error: 'Failed to generate PDF report.' }, { status: 500 });
  }
}

        onComplete(res, finalAnswers.map((a) => a ?? 0));
      }
    },
    [onComplete]
  );

  const handleStart = (e: React.FormEvent) => {
    e.preventDefault();
    if (!student.name.trim() || !student.email.trim() || !student.phone.trim()) {
      alert('Please fill in your Name, Email, and Mobile Number.');
      return;
    }

    setQi(0);
    setScreen('quiz');

  };

  const handleSimulate = () => {

      name: student.name.trim() || 'Aarav Sharma',
      email: student.email.trim() || 'aarav.sharma@example.com',

      3,4,4,3, 2,3,2,2, 2,2,3,2, 3,2,3,2, 3,4,3,3, 2,3,2,3, 4,5,4,5, 5,5,4,4,
      3,4,3,3, 3,4,3,4, 4,5,4,5,
    ];
    const simAnswers = QUESTIONS.map((_, idx) => pattern[idx % pattern.length]);
    setStudent(simStudent);
    setAnswers(simAnswers);
    finalizeReport(simStudent, simAnswers);
  };

    (optionIdx: number) => {
      if (busy || qi >= QUESTIONS.length) return;
      setBusy(true);

      setAnswers(nextAnswers);

      setTimeout(() => {
        setBusy(false);
        if (qi + 1 >= QUESTIONS.length) {
          finalizeReport(student, nextAnswers);
        } else {
          setQi((prev) => prev + 1);
        }

    },

    if (busy || qi === 0) return;
    setQi((prev) => prev - 1);

  useEffect(() => {
    if (screen !== 'quiz') return;
    const onKeyDown = (e: KeyboardEvent) => {
      if ((e.ctrlKey || e.metaKey) && ['c', 'x', 'u', 's', 'a'].includes(e.key.toLowerCase())) {
        e.preventDefault();
        return;
      }
      if (e.key >= '1' && e.key <= '5' && qi < QUESTIONS.length) {
        handlePick(Number(e.key) - 1);
      } else if (e.key === 'ArrowLeft' || e.key === 'Backspace') {
        e.preventDefault();
        handleBack();
      }
    };
    window.addEventListener('keydown', onKeyDown);
    return () => window.removeEventListener('keydown', onKeyDown);
  }, [screen, qi, handlePick, handleBack]);

        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ student, answers, branding }),
      });
      if (!res.ok) throw new Error('PDF generation failed');
      const blob = await res.blob();
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `Commerce_Career_Report_${student.name.replace(/\s+/g, '_')}.pdf`;
      a.click();
      URL.revokeObjectURL(url);
    } catch {
      alert('Could not download PDF. Please ensure /api/reports/commerce-pdf is running.');

      setDownloadingPdf(false);
    }
  };

  // ==================== SCREEN 1: REGISTRATION ====================
  if (screen === 'form') {
    return (
      <div className="min-h-screen bg-[#FAF5EC] py-10 px-4 text-slate-800">
        <form
          onSubmit={handleStart}

          <h1 className="font-serif text-3xl md:text-4xl font-bold text-slate-900">
            Commerce Career{' '}
            <span className="bg-gradient-to-r from-blue-600 via-sky-500 to-emerald-500 bg-clip-text text-transparent">
              Selector
            </span>
          </h1>
          <div className="w-20 h-1.5 rounded-full bg-gradient-to-r from-blue-600 to-emerald-500 my-3" />
          <p className="text-slate-500 mb-5">
            Discover which Commerce career cluster — financial or non-financial — matches your interests best.
          </p>

          <div className="flex flex-wrap gap-2 mb-6">
            {['76 Questions', '~20 Minutes', '19 Career Clusters', '10-Page A4 PDF Report'].map((chip) => (
              <span
                key={chip}
                className="bg-blue-50 text-blue-700 border border-blue-200 text-xs font-semibold px-3 py-1 rounded-full"

                {chip}

            <strong>Strongly Agree</strong>. Selecting an option automatically saves and advances.

              <input
                type="text"
                required
                value={student.name}
                onChange={(e) => setStudent({ ...student, name: e.target.value })}
                placeholder="e.g. Aarav Sharma"

              />
            </div>

              <label className="block text-sm font-semibold text-slate-700 mb-1">Email Address</label>

                type="email"
                required
                value={student.email}
                onChange={(e) => setStudent({ ...student, email: e.target.value })}

                required
                value={student.phone}

    return (

        className="min-h-screen bg-[#FAF5EC] py-8 px-4 select-none"

      >
        <div className="max-w-3xl mx-auto">

                Rate Yourself

                Question {qi + 1} of {QUESTIONS.length}

              {q.text}

                  >
                    <span
                      className={`w-8 h-8 rounded-full flex items-center justify-center text-sm font-bold shrink-0 ${
                        selected ? 'bg-white/25 text-white' : 'bg-[#EDE7DA] text-slate-700'

                    >
                      {idx + 1}
                    </span>
                    <span>{label}</span>
                  </button>

              <button
                type="button"
                onClick={handleBack}
                disabled={qi === 0}
                className="inline-flex items-center gap-1.5 px-4 py-2 rounded-xl border-2 border-[#E7E0D2] text-slate-600 font-semibold text-sm hover:border-blue-600 hover:text-blue-600 disabled:invisible"
              >
                <ArrowLeft className="w-4 h-4" /> Previous
              </button>
              <span className="text-xs font-semibold text-slate-400">
                Keys 1–5 to answer · Auto-advances on selection
              </span>
            </div>
          </div>
        </div>
      </div>
    );
  }

  // ==================== SCREEN 3: COLORFUL INTERACTIVE REPORT ====================
  if (!report) return null;

  return (
    <div className="min-h-screen bg-[#FAF5EC] py-8 px-4 text-slate-800">
      <div className="max-w-5xl mx-auto space-y-6">
        {/* Sticky Download Toolbar */}
        <div className="sticky top-3 z-30 flex flex-wrap items-center justify-between gap-4 bg-white/95 backdrop-blur-md border border-[#E7E0D2] rounded-2xl px-5 py-3.5 shadow-lg">
          <div className="flex items-center gap-3">
            <ShieldCheck className="w-6 h-6 text-blue-600" />
            <div>
              <div className="font-bold text-slate-900 text-sm md:text-base">
                {branding?.clientName || 'Commerce Career Selector'} — Complete Report
              </div>
              <div className="text-xs text-slate-500">
                10-Page A4 PDF generated via backend <code className="font-mono">pdf-lib</code>
              </div>
            </div>
          </div>
          <button
            type="button"
            onClick={handleDownloadPdf}
            disabled={downloadingPdf}
            className="inline-flex items-center gap-2 px-5 py-2.5 rounded-xl bg-gradient-to-r from-blue-600 to-sky-500 text-white font-semibold text-sm shadow-md hover:opacity-95 disabled:opacity-60 transition"
          >
            <Download className="w-4 h-4" />
            {downloadingPdf ? 'Generating A4 PDF...' : 'Download 10-Page A4 PDF'}
          </button>
        </div>

        {/* Section 1: Cover & Executive KPIs */}
        <section className="bg-white rounded-2xl border border-[#E7E0D2] p-6 md:p-8 shadow-sm space-y-6">
          <div className="flex flex-wrap items-center justify-between gap-4 border-b-2 border-blue-600 pb-4">
            <div className="flex items-center gap-4">
              {branding?.logoUrl && (
                <img src={branding.logoUrl} alt="Client Logo" className="max-h-16 w-auto object-contain" />
              )}
              <div>
                <div className="font-serif text-2xl font-bold text-slate-900">
                  {branding?.clientName || 'Commerce Career Selector'}
                </div>
                <div className="text-xs font-bold uppercase tracking-wider text-blue-600">
                  Psychometric Career Assessment &amp; Counseling Report
                </div>
              </div>
            </div>
            <div className="text-right text-xs font-semibold text-slate-500">
              <div>REPORT ID: {report.reportId}</div>
              <div className="text-blue-600">VALIDATED ASSESSMENT</div>
            </div>
          </div>

          <div className="bg-gradient-to-r from-blue-50 to-slate-50 border border-blue-200 border-l-4 border-l-blue-600 rounded-xl p-5">
            <h1 className="font-serif text-2xl md:text-3xl font-bold text-slate-900 mb-1">{report.student.name}</h1>
            <div className="flex flex-wrap gap-x-6 gap-y-1 text-xs font-semibold text-slate-500">
              <span>DATE: <strong className="text-slate-900">{report.dateFormatted}</strong></span>
              <span>PHONE: <strong className="text-slate-900">{report.student.phone}</strong></span>
              <span>EMAIL: <strong className="text-slate-900">{report.student.email}</strong></span>
              <span>COMPLETED: <strong className="text-slate-900">{report.answeredCount}/{report.totalQuestions}</strong></span>
            </div>

          <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
            <div className="bg-[#FAF8F2] border border-[#E7E0D2] border-t-4 border-t-blue-600 rounded-xl p-4 text-center">
              <div className="font-serif text-2xl font-bold text-blue-600">{report.preferredDomain.split(' ')[0]}</div>
              <div className="text-[11px] font-bold uppercase text-slate-500 mt-1">Preferred Domain</div>
              <div className="text-xs text-slate-600">{Math.max(report.financialPct, report.nonFinancialPct)}% Share</div>
            </div>
            <div className="bg-[#FAF8F2] border border-[#E7E0D2] border-t-4 border-t-emerald-500 rounded-xl p-4 text-center">
              <div className="font-serif text-2xl font-bold text-emerald-600">{report.top3[0].percentage}%</div>

              <div className="text-xs text-slate-600 truncate">{report.top3[0].cluster}</div>

              <div className="font-serif text-2xl font-bold text-indigo-600">{report.overallIndexPct}%</div>
              <div className="text-[11px] font-bold uppercase text-slate-500 mt-1">Overall Index</div>
              <div className="text-xs text-slate-600">{report.totalRawScore} / 380 Pts</div>
            </div>
            <div className="bg-[#FAF8F2] border border-[#E7E0D2] border-t-4 border-t-amber-500 rounded-xl p-4 text-center">
              <div className="font-serif text-2xl font-bold text-amber-600">{report.highCount} / 19</div>

                      </span>
                    </td>
                    <td className="p-3 border border-[#E7E0D2] text-slate-700">{c.roles.join(', ')}</td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>

          <div className="bg-blue-50 border border-blue-200 border-l-4 border-l-blue-600 rounded-xl p-5 flex gap-3 items-start">
            <CheckCircle2 className="w-6 h-6 text-blue-600 shrink-0 mt-0.5" />
            <div>
              <h3 className="font-bold text-blue-800 text-sm mb-1">Actions &amp; Next Steps for {report.student.name}</h3>
              <p className="text-sm text-slate-700">
                Prioritize your High-level clusters — <strong>{report.top3.map((c) => c.cluster).join(', ')}</strong>.
                Download the 10-page A4 PDF above to review your full radar geometry, domain breakdown, and 3-step counselor roadmap.
              </p>
            </div>
          </div>
        </section>
      </div>
    </div>
  );
}
Key Architectural Advantages
Single Source of Truth: calculateCommerceResults() in scoring.ts powers both the React UI and the pdf-lib backend, guaranteeing identical scores, rankings, and High/Medium/Low tiers.

Zero Browser Print Issues: Because pdf-lib builds the 10-page A4 PDF directly on the backend with vector charts and exact point coordinates, there is zero risk of browser print-dialog clipping, blank spillover pages, or missing Canvas graphics.

Lightweight on AWS EC2: Unlike Puppeteer/Playwright, pdf-lib requires no headless Chrome binary and uses minimal RAM, keeping your server costs at zero.
